import re

phase = '.planning/phases/102-institutional-joint-staff-teams'

duty_pattern = r'^\| ((?:CG|SS|J[1-9])-(?:[A-Z]+)?[0-9]{2})\b'
all_ids = set()
for name, expected in zip(sources, (73, 38, 38), strict=True):
    path = f'.planning/research/staff-teams/{name}'
    reviewed = git('show', f'360515a60:{path}')
    assert reviewed == git('show', f'{base}:{path}'), path
    ids = re.findall(duty_pattern, reviewed, re.M)
    assert len(ids) == expected and len(set(ids)) == expected and not all_ids.intersection(ids), path
    all_ids.update(ids)

totals = Counter(duty.split('-')[0] for duty in all_ids)
assert len(all_ids) == 149 and totals['CG'] == 12 and totals['SS'] == 20
assert sum(totals[f'J{i}'] for i in range(1, 10)) == 117
