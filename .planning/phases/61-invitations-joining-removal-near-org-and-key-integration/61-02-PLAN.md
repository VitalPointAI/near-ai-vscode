```typescript
export const NEAR_SPEND_CURRENCY = 'USD';
export interface NearAuth { accessToken(): Promise<string>; refreshAfterUnauthorized(): Promise<string | undefined> }
export type NearOrgErrorCode = 'unauthorized' | 'forbidden' | 'not-found' | 'conflict' | 'bad-request' | 'rate-limited' | 'server' | 'network' | 'bad-response' | 'limit-mismatch';
export class NearOrgError extends Error { readonly status: number; readonly code: NearOrgErrorCode }
export interface NearOrganization { id: string; name: string; role: 'owner' | 'admin' | 'member' }
export interface NearWorkspace { id: string; name: string; organizationId: string }
export interface NearOrgMember { userId: string; role: 'owner' | 'admin' | 'member'; email?: string }
export interface NearInvitation { id: string; organizationId: string; email: string; role: string; status: string; expiresAt?: string; organizationName?: string }
export type InviteOutcome = { ok: true; alreadyMember: boolean } | { ok: false; reason: string };
export interface NearKey { id: string; name: string; workspaceId: string; createdByUserId?: string; isActive: boolean; spendLimitNanoUsd: number | null; usageNanoUsd: number | null }
export interface CreatedNearKey { key: NearKey; secret: Buffer }                // secret is the sk- key; the caller wipes it
export interface NearUsageEntry { id: string; model: string; totalCostNanoUsd: number; createdAt: string }
export type CheckedKey = { status: 'valid' | 'capped'; organizationId: string; workspaceId?: string; apiKeyId: string } | { status: 'invalid' | 'unavailable'; reason: string };
export class NearOrgClient {
  constructor(options: { baseUrl: string; auth: NearAuth; fetch?: FetchLike; sleep?: (ms: number) => Promise<void>; maxRetries?: number });
  listOrganizations(): Promise<NearOrganization[]>;
  listWorkspaces(organizationId: string): Promise<NearWorkspace[]>;
  listMembers(organizationId: string): Promise<NearOrgMember[]>;
  removeMember(organizationId: string, userId: string): Promise<'removed' | 'missing'>;
  inviteByEmail(organizationId: string, email: string, role: 'admin' | 'member'): Promise<InviteOutcome>;
  listInvitations(organizationId: string, status?: 'pending' | 'accepted' | 'declined' | 'expired'): Promise<NearInvitation[]>;
  cancelInvitation(organizationId: string, invitationId: string): Promise<'cancelled' | 'not-pending'>;
  myInvitations(): Promise<NearInvitation[]>;
  acceptInvitation(invitationId: string): Promise<'accepted' | 'already-member'>;
  acceptInvitationToken(token: string): Promise<'accepted' | 'already-member'>;
  createKey(workspaceId: string, name: string, spendLimitNanoUsd: number): Promise<CreatedNearKey>;
  listKeys(workspaceId: string): Promise<NearKey[]>;
  findKey(workspaceId: string, keyId: string): Promise<NearKey | undefined>;
  setSpendLimit(workspaceId: string, keyId: string, nanoUsd: number): Promise<NearKey | 'missing'>;
  revokeKey(workspaceId: string, keyId: string): Promise<'revoked' | 'missing'>;
  keyUsageHistory(workspaceId: string, keyId: string, page?: { limit: number; offset: number }): Promise<{ entries: NearUsageEntry[]; total: number } | 'not-offered'>;
  currentUser(): Promise<{ id: string; username: string; email?: string; organizationIds: string[] }>;
}
export function checkApiKey(baseUrl: string, apiKey: Buffer | string, fetch?: FetchLike): Promise<CheckedKey>;
```
