```typescript
export const TEAM_ACTION_IDS = ['turn-on', 'join', 'share', 'invite', 'members', 'board', 'link', 'check'] as const;
export type TeamActionId = (typeof TEAM_ACTION_IDS)[number];
export const TEAM_ACTION_COMMANDS: Readonly<Record<TeamActionId, string>>;
// turn-on nearAi.team.enable, join nearAi.team.join, share nearAi.team.share, invite nearAi.team.invite,
// members nearAi.team.manageMembers, board nearAi.team.openBoard, link nearAi.team.linkNearOrganization, check nearAi.team.verifyLog
export function isTeamActionId(value: unknown): value is TeamActionId;
export interface TeamBoardSummary { ready: number; blocked: number; mine: number }       // filled by Phase 62's board fix
export type SharedView = 'yes' | 'no' | 'not-reachable-by-teammates' | 'unknown';
export interface TeamSectionInput {
  folderOpen: boolean;
  team?: {
    name: string;                         // display-safe text, already passed through displayText by the host
    myRole?: Role;                        // undefined: this computer is not on the team
    pendingRole?: Role;                   // joined, waiting for a lead
    removed: boolean;
    canManage: boolean;                   // owner or lead
    waiting: number;                      // memberAttention count, managers only
    pendingInvites: number;               // invitations not yet used
    otherMembers: number;                 // joined people other than me
    linked: boolean;                      // organization link in team.json
    connection: 'live' | 'syncing' | 'offline' | 'paused';
    connectionText?: string;              // boardStatus text for offline or paused
    shared?: SharedView;                  // managers only; undefined while not checked yet
  };
  board?: TeamBoardSummary;
}
export interface TeamSectionAction { id: TeamActionId; label: string; primary: boolean; badge?: string; hint?: string }
export interface TeamSectionModel { heading: string; status: string; compact: boolean; actions: TeamSectionAction[]; board?: TeamBoardSummary & { text: string } }
export function teamSectionModel(input: TeamSectionInput): TeamSectionModel | undefined;   // undefined when no folder is open
export interface TeamSetupStep { id: 'team.turn-on' | 'team.share' | 'team.invite' | 'team.link'; title: string; done: boolean; action: TeamActionId; optional: boolean }
export function teamSetupSteps(input: TeamSectionInput): TeamSetupStep[];
```
