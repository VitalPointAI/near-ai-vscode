export type NotReadyReason = 'not-unclaimed' | 'held' | 'blocked' | 'no-estimate' | 'no-scope' | 'dependency-open' | 'dependency-missing' | 'phase-held' | 'phase-dependency-open';
export function isCard(item: WorkItem): boolean;                       // kind plan, or origin board
export function readiness(items: ReadonlyMap<string, WorkItem>, id: string): { ready: boolean; reasons: NotReadyReason[] };
export function effectiveColumn(items: ReadonlyMap<string, WorkItem>, item: WorkItem): BoardColumn;
export function progressOf(items: ReadonlyMap<string, WorkItem>, groupId: string): { percent: number; done: number; total: number };
export function countReady(items: ReadonlyMap<string, WorkItem>): number;
// itemHistory.ts
export const MAX_HISTORY_PER_ITEM = 50;
export interface HistoryEntry { eventId: string; type: string; author: string; at: string; reason?: string; changed?: string[] }
export class ItemHistory { record(entries: readonly CommittedEvent[]): void; forItem(itemId: string): HistoryEntry[]; }   // newest first
